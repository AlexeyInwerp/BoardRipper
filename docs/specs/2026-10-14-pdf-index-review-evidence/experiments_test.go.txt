func TestZZPoolKill(t *testing.T) {
	data, _ := os.ReadFile("../../frontend/tests/fixtures/many-page-text.pdf")
	t0 := time.Now()
	eng, err := NewEngine(2)
	if err != nil { t.Fatal(err) }
	defer eng.Close()
	fmt.Printf("NewEngine(2): %v\n", time.Since(t0))
	// baseline normal extracts (instance create+close per file)
	for i := 0; i < 5; i++ {
		s := time.Now()
		p, err := eng.ExtractFile(data)
		fmt.Printf("normal extract %d: %v pages=%d err=%v\n", i, time.Since(s), len(p), err)
	}
	// direct kill rounds
	for round := 1; round <= 6; round++ {
		a, e1 := eng.pool.GetInstance(5 * time.Second)
		b, e2 := eng.pool.GetInstance(5 * time.Second)
		if e1 != nil || e2 != nil { fmt.Printf("round %d get fail %v %v\n", round, e1, e2); break }
		a.Kill(); b.Kill()
		s := time.Now()
		c, e3 := eng.pool.GetInstance(30 * time.Second)
		fmt.Printf("kill round %d (2 killed): third GetInstance %v err=%v\n", round, time.Since(s), e3)
		if c != nil {
			var pr atomic.Int32
			s2 := time.Now()
			pg, e4 := extract(c, data, &pr)
			fmt.Printf("   extract on replacement: %v pages=%d err=%v\n", time.Since(s2), len(pg), e4)
			c.Close()
		}
	}
	// ExtractFile with tiny timeout
	eng.perFileTimeout = 1 * time.Millisecond
	for i := 0; i < 8; i++ {
		s := time.Now()
		_, err := eng.ExtractFile(data)
		fmt.Printf("ExtractFile timeout %d: %v err=%v\n", i, time.Since(s), err)
	}
	eng.perFileTimeout = 2 * time.Minute
	s := time.Now()
	p, err := eng.ExtractFile(data)
	fmt.Printf("after timeouts, normal extract: %v pages=%d err=%v\n", time.Since(s), len(p), err)
}

func TestZZConcurrent(t *testing.T) {
	db, err := Open(zzDir + "/pdfindex.db")
	if err != nil { t.Fatal(err) }
	defer db.Close()
	wal := zzDir + "/pdfindex.db-wal"
	stop := make(chan struct{})
	var writes atomic.Int64
	go func() {
		r := rand.New(rand.NewSource(7))
		for f := int64(1); ; f = f%2000 + 1 {
			select { case <-stop: return; default: }
			pages := make([]Page, 100)
			for i := range pages { pages[i] = Page{Num: i + 1, Text: zzGen(r)} }
			db.ReplacePages(f, pages)
			writes.Add(1)
		}
	}()
	held := make(chan struct{})
	go func() {
		first := true
		db.SearchPagesStream("PP3V3", nil, 1000, func(h SearchHit) error {
			if first { first = false; fmt.Println("HELD cursor first row at", time.Now().Format("15:04:05")); time.Sleep(30 * time.Second) }
			return nil
		})
		fmt.Println("HELD cursor released at", time.Now().Format("15:04:05"))
		close(held)
	}()
	t0 := time.Now()
	qs := []string{"10UF 25V", "PP3V3", "C", "U12", "SMC_RESET_L"}
	var lat = map[string][]time.Duration{}
	nextWal := time.Time{}
	for time.Since(t0) < 60*time.Second {
		for _, q := range qs {
			s := time.Now()
			db.SearchPages(q, nil, 1000)
			phase := "held"; if time.Since(t0) > 31*time.Second { phase = "after" }
			lat[q+"/"+phase] = append(lat[q+"/"+phase], time.Since(s))
			if time.Now().After(nextWal) {
				fmt.Printf("t=%2.0fs wal=%d MB db=%d MB writes(files)=%d\n", time.Since(t0).Seconds(), zzSize(wal)>>20, zzSize(zzDir+"/pdfindex.db")>>20, writes.Load())
				nextWal = time.Now().Add(5 * time.Second)
			}
		}
	}
	close(stop)
	for k, v := range lat {
		sort.Slice(v, func(i, j int) bool { return v[i] < v[j] })
		fmt.Printf("LAT %-24s n=%d min=%v med=%v max=%v\n", k, len(v), v[0], v[len(v)/2], v[len(v)-1])
	}
	<-held
}

func TestZZLaneB(t *testing.T) {
	db := openTestDB(t)
	const iters = 300
	src := &zzSrc{files: map[int64]PdfFile{}}
	nextID := int64(1000)
	var lost, total, stuckPriority int
	for it := 0; it < iters; it++ {
		// one pending file in the list so a sweep happens
		base := nextID; nextID++
		src.mu.Lock(); src.files[base] = PdfFile{base, fmt.Sprintf("b%d.pdf", base)}; src.listIDs = []int64{base}; src.mu.Unlock()
		ix := NewIndexer(db, &zzExt{}, src, func() []string { return nil }, 2)
		var ids []int64
		var wg sync.WaitGroup
		wg.Add(1)
		ix.Run()
		go func() {
			defer wg.Done()
			for k := 0; k < 40 && ix.Progress().Running; k++ {
				id := nextID; nextID++
				src.mu.Lock(); src.files[id] = PdfFile{id, fmt.Sprintf("s%d.pdf", id)}; src.mu.Unlock()
				ix.Submit(id)
				ids = append(ids, id)
			}
		}()
		wg.Wait()
		for d := 0; d < 200 && ix.Progress().Running; d++ { time.Sleep(10 * time.Millisecond) }
		time.Sleep(20 * time.Millisecond)
		for _, id := range ids {
			st, _ := db.Status(id)
			total++
			if st.Status != "indexed" { lost++ }
		}
		stuckPriority += len(ix.priority)
	}
	fmt.Printf("T4b iterations=%d submitted=%d notIndexed(after Running==false)=%d leftInPriorityChan=%d\n", iters, total, lost, stuckPriority)
}

func TestZZWatchdog(t *testing.T) {
	for _, withSecondSweep := range []bool{false, true} {
		db := openTestDB(t)
		src := &zzSrc{files: map[int64]PdfFile{1: {1, "a.pdf"}}, listIDs: []int64{1}}
		ext := &zzExt{delay: 3 * time.Second}
		ix := NewIndexer(db, ext, src, func() []string { return nil }, 2)
		stop := ix.StartWatchdog(200*time.Millisecond, 1)
		ix.Run()
		var seen []string
		t0 := time.Now()
		second := false
		for time.Since(t0) < 5*time.Second {
			st, _ := db.Status(1)
			if len(seen) == 0 || seen[len(seen)-1] != st.Status { seen = append(seen, st.Status); fmt.Printf("  t=%.1fs status=%q\n", time.Since(t0).Seconds(), st.Status) }
			if withSecondSweep && !second && st.Status == "pending" {
				second = true
				// a periodic/manual sweep arrives while job 1 is still live
				fmt.Printf("  t=%.1fs Submit(1) err=%v (sweep Running=%v)\n", time.Since(t0).Seconds(), ix.Submit(1), ix.Progress().Running)
			}
			time.Sleep(50 * time.Millisecond)
		}
		close(stop)
		v, _ := ext.calls.Load("text a.pdf")
		fmt.Printf("T5 secondSweep=%v statuses=%v extractCalls=%d\n", withSecondSweep, seen, v.(*atomic.Int32).Load())
	}
}

func zzQ0(db *DB, fts string, limit int, snip bool) (int, error) {
	sn := ", snippet(pdf_text, 0, '<b>', '</b>', '...', 32) AS snip"
	sel := "r.snip"
	if !snip { sn = ""; sel = "''" }
	q := `WITH ranked AS (
		SELECT pdf_text.rowid, rank` + sn + `
		FROM pdf_text WHERE pdf_text MATCH ? ORDER BY rank LIMIT ?
	)
	SELECT p.file_id, p.page_num, ` + sel + `
	FROM ranked r JOIN pdf_pages p ON p.rowid = r.rowid
	ORDER BY r.rank`
	rows, err := db.reader.Query(q, fts, limit)
	if err != nil { return 0, err }
	defer rows.Close()
	n := 0
	for rows.Next() { var a, b int64; var c string; rows.Scan(&a, &b, &c); n++ }
	return n, rows.Err()
}

	if err != nil { return 0, err }
	for r3.Next() { var a, b int64; r3.Scan(&a, &b) }
	r3.Close()
	return n, nil
}

				err := db.reader.QueryRow(`SELECT COUNT(*) FROM pdf_text WHERE pdf_text MATCH ?`, fts).Scan(&n)
				return n, err }},
			{"Q6 per-file", func() (int, error) {
				r, err := db.reader.Query(`SELECT p.file_id, MIN(p.page_num) FROM pdf_text t JOIN pdf_pages p ON p.rowid=t.rowid WHERE pdf_text MATCH ? GROUP BY p.file_id LIMIT 200`, fts)
				if err != nil { return 0, err }
				defer r.Close(); n := 0
				for r.Next() { var a, b int64; r.Scan(&a, &b); n++ }
				return n, r.Err() }},
		}
		for _, x := range vs {
			if os.Getenv("ZZ_SKIPQ1") != "" && (strings.HasPrefix(x.name, "Q1") || strings.HasPrefix(x.name, "Q3") || strings.HasPrefix(x.name, "Q6")) { continue }
			if strings.HasPrefix(x.name, "Q1") && (q == "C" || q == "1") { fmt.Printf("RES %-10q %-14s SKIPPED (did not finish in >5 min)\n", q, x.name); continue }
			if strings.HasPrefix(x.name, "Q1") {
				s := time.Now(); n, err := x.f(); fmt.Printf("RES %-10q %-14s n=%d err=%v single-run=%v\n", q, x.name, n, err, time.Since(s).Round(time.Millisecond)); continue
			}
			s := time.Now(); n, err := x.f(); cold := time.Since(s)
			s = time.Now(); x.f(); w1 := time.Since(s)
			s = time.Now(); x.f(); w2 := time.Since(s)
			fmt.Printf("RES %-10q %-14s n=%d err=%v first=%v warm=%v,%v\n", q, x.name, n, err, cold.Round(time.Millisecond), w1.Round(time.Millisecond), w2.Round(time.Millisecond))
		}
	}
}
